{{/* param-table: renders a parameter table from data/params//.yaml. Optional columns appear only when a row uses them. The optional group= filter selects one group out of a file that holds several. The global pages use it, since all of a source's params live in one file and each row carries its group. Per-scenario tables omit it and render every row. Filtering happens before column detection on purpose: a group with no possible_values must not inherit that column from a sibling group. */}} {{- $scenario := .Get "scenario" -}} {{- $source := .Get "source" -}} {{- $group := .Get "group" -}} {{- $data := false -}} {{- with site.Data.params -}}{{- $data = index . $scenario $source -}}{{- end -}} {{- if or (not $data) (not $data.params) -}} {{- errorf "param-table: no data for scenario=%q source=%q (expected data/params/%s/%s.yaml)" $scenario $source $scenario $source -}} {{- else -}} {{- $rows := $data.params -}} {{- with $group -}}{{- $rows = where $rows "group" . -}}{{- end -}} {{- if and $group (not $rows) -}} {{- errorf "param-table: group %q has no params in data/params/%s/%s.yaml" $group $scenario $source -}} {{- else -}} {{- $showType := false -}}{{- $showValues := false -}}{{- $showDefault := false -}}{{- $showRequired := false -}} {{- range $rows -}} {{- if isset . "type" }}{{ $showType = true }}{{ end -}} {{- if isset . "possible_values" }}{{ $showValues = true }}{{ end -}} {{- if isset . "default" }}{{ $showDefault = true }}{{ end -}} {{- if isset . "required" }}{{ $showRequired = true }}{{ end -}} {{- end -}} {{- if $showType }}{{ end }} {{- if $showValues }}{{ end }} {{- if $showDefault }}{{ end }} {{- if $showRequired }}{{ end }} {{- range $rows }} {{- if $showType }}{{ end }} {{- if $showValues }}{{ end }} {{- if $showDefault }}{{ end }} {{- if $showRequired }}{{ end }} {{- end }}
